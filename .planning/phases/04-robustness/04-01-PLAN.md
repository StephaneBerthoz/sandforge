```typescript
import { RetryStrategy, type RetryConfig, type RetryResult } from './RetryStrategy.js';
import { ErrorClassifier, type ClassifiedError } from './ErrorClassifier.js';

interface RetryableOperationOptions {
  retryConfig?: Partial<RetryConfig>;
  onRetry?: (attempt: number, error: ClassifiedError, delay: number) => void;
}

export class RetryableOperation {
  private strategy: RetryStrategy;
  private classifier: ErrorClassifier;

  constructor(private options: RetryableOperationOptions = {}) {
    this.strategy = new RetryStrategy(options.retryConfig);
    this.classifier = new ErrorClassifier();
  }

  /** Execute with automatic retry for transient errors */
  async execute<T>(fn: () => Promise<T>): Promise<RetryResult<T>> {
    return this.strategy.execute(async () => {
      try {
        return await fn();
      } catch (err) {
        const classified = this.classifier.classify(err as any);
        if (!classified.classification.retryable) {
          throw err; // Don't retry non-retryable errors
        }
        this.options.onRetry?.(0, classified, classified.classification.delay ?? 1000);
        throw err; // Let RetryStrategy handle the retry
      }
    });
  }

  /** Execute batch with smart retry: retry only failed records for record-level errors */
  async executeBatch<T>(
    records: T[],
    fn: (batch: T[]) => Promise<{ successes: T[]; failures: { record: T; error: any }[] }>,
  ): Promise<{ successes: T[]; failures: { record: T; error: any }[] }> {
    let remaining = [...records];
    const allSuccesses: T[] = [];
    const allFailures: { record: T; error: any }[] = [];
    let attempts = 0;
    const maxAttempts = this.options.retryConfig?.maxRetries ?? 3;

    while (remaining.length > 0 && attempts <= maxAttempts) {
      const result = await fn(remaining);
      allSuccesses.push(...result.successes);

      // Classify failures
      const retryable: T[] = [];
      for (const f of result.failures) {
        const classified = this.classifier.classify(f.error);
        if (classified.classification.retryable && attempts < maxAttempts) {
          retryable.push(f.record);
        } else {
          allFailures.push(f);
        }
      }

      remaining = retryable;
      if (remaining.length > 0) {
        const delay = (this.options.retryConfig?.initialDelay ?? 1000) * Math.pow(2, attempts);
        await new Promise(r => setTimeout(r, Math.min(delay, 30000)));
        attempts++;
      }
    }

    return { successes: allSuccesses, failures: allFailures };
  }
}
```

Create TimeoutManager using AbortController pattern:

```typescript
export class TimeoutError extends Error {
  constructor(public readonly operation: string, public readonly timeoutMs: number) {
    super(`Operation "${operation}" timed out after ${timeoutMs}ms`);
    this.name = 'TimeoutError';
  }
}

export class TimeoutManager {
  constructor(private defaultTimeoutMs: number = 30_000) {}

  /** Wrap an async operation with a timeout */
  async withTimeout<T>(
    operation: string,
    fn: (signal: AbortSignal) => Promise<T>,
    timeoutMs?: number,
  ): Promise<T> {
    const timeout = timeoutMs ?? this.defaultTimeoutMs;
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeout);

    try {
      const result = await Promise.race([
        fn(controller.signal),
        new Promise<never>((_, reject) => {
          controller.signal.addEventListener('abort', () => {
            reject(new TimeoutError(operation, timeout));
          });
        }),
      ]);
      return result;
    } finally {
      clearTimeout(timer);
    }
  }
}
```

Tests:
- withTimeout() returns result when operation completes before timeout
- withTimeout() throws TimeoutError when operation exceeds timeout
- TimeoutError has correct operation name and timeoutMs
- AbortSignal is aborted when timeout fires
- Uses custom timeout when provided, falls back to default

import { BulkApiManager, type BulkJobInfo, type BulkJobOptions } from './BulkApiManager.js';
import { SF_LIMITS } from '@sandforge/shared';

export interface BulkExecutionResult {
  totalRecords: number;
  successCount: number;
  failureCount: number;
  failures: { recordIndex: number; error: string }[];
  jobId: string;
  usedBulkApi: boolean;
}

export interface BulkApiExecutorDeps {
  connection: any; // jsforce.Connection
  bulkManager: BulkApiManager;
  onProgress?: (processed: number, total: number) => void;
}

export class BulkApiExecutor {
  private threshold: number;

  constructor(threshold: number = SF_LIMITS.REST_API_BATCH_SIZE) {
    this.threshold = threshold;
  }

  /** Should we use Bulk API for this operation? */
  shouldUseBulkApi(recordCount: number): boolean {
    return recordCount > this.threshold;
  }

  /** Execute insert/upsert/update/delete via Bulk API 2.0 */
  async executeBulk(
    deps: BulkApiExecutorDeps,
    objectName: string,
    operation: BulkJobOptions['operation'],
    records: Record<string, unknown>[],
    externalIdField?: string,
  ): Promise<BulkExecutionResult> {
    if (!deps.bulkManager.canStartNewJob()) {
      throw new Error('Maximum concurrent bulk jobs reached');
    }

    // Create Bulk 2.0 job via jsforce
    const job = deps.connection.bulk2.createJob({
      operation,
      object: objectName,
      ...(externalIdField && { externalIdFieldName: externalIdField }),
    });

    const jobInfo: BulkJobInfo = {
      id: job.id ?? `bulk-${Date.now()}`,
      operation,
      object: objectName,
      state: 'UploadComplete',
      numberRecordsProcessed: 0,
      numberRecordsFailed: 0,
      totalProcessingTime: 0,
      createdDate: new Date().toISOString(),
    };
    deps.bulkManager.registerJob(jobInfo);

    // Upload, close, poll for results
    await job.open();
    await job.uploadData(records);
    await job.close();

    // Poll until complete
    let status = await job.check();
    while (status.state === 'InProgress' || status.state === 'UploadComplete') {
      deps.bulkManager.updateJobState(jobInfo.id, status.state);
      deps.onProgress?.(status.numberRecordsProcessed ?? 0, records.length);
      await new Promise(r => setTimeout(r, 5000));
      status = await job.check();
    }

    // Get results
    const results = await job.getAllResults();
    const failures = results
      .filter((r: any) => !r.success)
      .map((r: any, i: number) => ({ recordIndex: i, error: r.errors?.join(', ') ?? 'Unknown error' }));

    deps.bulkManager.updateJobState(jobInfo.id, status.state === 'JobComplete' ? 'JobComplete' : 'Failed');
    deps.bulkManager.updateJobCounts(jobInfo.id, status.numberRecordsProcessed ?? 0, failures.length);

    return {
      totalRecords: records.length,
      successCount: records.length - failures.length,
      failureCount: failures.length,
      failures,
      jobId: jobInfo.id,
      usedBulkApi: true,
    };
  }
}

export const RobustnessConfigSchema = z.object({
  timeouts: z.object({
    describeGlobal: z.number().min(5000).max(120_000).default(30_000),
    describe: z.number().min(5000).max(60_000).default(15_000),
    crudBatch: z.number().min(10_000).max(300_000).default(60_000),
    bulkJob: z.number().min(60_000).max(600_000).default(300_000),
  }).default({}),
  retry: z.object({
    maxRetries: z.number().min(0).max(10).default(3),
    initialDelay: z.number().min(100).max(30_000).default(1000),
    maxDelay: z.number().min(1000).max(120_000).default(30_000),
    backoffMultiplier: z.number().min(1).max(5).default(2),

    pollIntervalMs: z.number().min(1000).max(30_000).default(5000),
    maxConcurrentJobs: z.number().min(1).max(100).default(5),
  }).default({}),
});

export type RobustnessConfig = z.infer<typeof RobustnessConfigSchema>;
