<action>
Create `.github/workflows/release.yml` for one-click Marketplace publishing:

```yaml
name: Release

on:
  workflow_dispatch:
    inputs:
      version:
        description: 'Version bump type'
        required: true
        type: choice
        options:
          - patch
          - minor
          - major

jobs:
  release:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    permissions:
      contents: write

    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
          token: ${{ secrets.GITHUB_TOKEN }}

      - uses: pnpm/action-setup@v4
        with:
          version: 9

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Configure git
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"

      - name: Bump version
        run: |
          bash scripts/bump-version.sh ${{ inputs.version }}
          NEW_VERSION=$(node -p "require('./package.json').version")
          echo "NEW_VERSION=$NEW_VERSION" >> $GITHUB_ENV

      - name: Update lockfile
        run: pnpm install --no-frozen-lockfile

      - name: Commit version bump
        run: |
          git add -A
          git commit -m "chore: release v${{ env.NEW_VERSION }}"
          git tag "v${{ env.NEW_VERSION }}"

      - name: Validate
        run: pnpm validate

      - name: Package VSIX
        run: pnpm package

      - name: Publish to Marketplace
        run: npx @vscode/vsce publish --packagePath sandforge.vsix
        env:
          VSCE_PAT: ${{ secrets.VSCE_PAT }}

      - name: Push commit and tag
        run: |
          git push origin master
          git push origin "v${{ env.NEW_VERSION }}"

      - name: Create GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          tag_name: v${{ env.NEW_VERSION }}
          name: SandForge v${{ env.NEW_VERSION }}
          files: sandforge.vsix
          generate_release_notes: true
```

Key design decisions:
- `workflow_dispatch` with version choice — no accidental releases
- Bump → validate → package → publish → push tag → GitHub release
- Commit and tag created before publish, pushed after successful publish
- GitHub Release with VSIX as downloadable asset
- `permissions: contents: write` for tag push and release creation
- Uses `VSCE_PAT` secret (user must configure in repo settings)
