Demoable: Playwright spec that mocks a 3-event drift stream and asserts the timeline + filter chips work end-to-end.

## Non-goals

/** Allowlist of comparison fields per metadata type (P-03.2). */
export const FIELD_ALLOWLIST: Record<string, readonly string[]> = {
  CustomField: ['type', 'length', 'picklistValues', 'required', 'externalId', 'unique', 'referenceTo'],
  PermissionSet: ['fieldPermissions', 'objectPermissions'],
  Profile: ['fieldPermissions', 'objectPermissions'],
};

/** Fields that change on every fetch — always strip. */
export const NOISE_FIELDS = new Set(['lastModifiedDate', 'lastModifiedById', 'systemModstamp', 'urls', 'attributes']);

export function canonicalizeField(raw: any): any {
  if (!raw) return raw;
  const out: any = {};
  for (const key of FIELD_ALLOWLIST.CustomField) {
    if (key in raw) out[key] = raw[key];
  }
  // Sort picklist values by `value` for stable diff
  if (Array.isArray(out.picklistValues)) {
    out.picklistValues = [...out.picklistValues].sort((a, b) => String(a.value).localeCompare(String(b.value)));
  }
  return out;
}

export function canonicalizePermissionSet(raw: any): any {
  if (!raw) return raw;
  const out: any = {};
  // Sort fieldPermissions by `field` so unstable ordering doesn't trigger spurious diffs
  if (Array.isArray(raw.fieldPermissions)) {
    out.fieldPermissions = [...raw.fieldPermissions].sort((a, b) => String(a.field).localeCompare(String(b.field)));
  }
  if (Array.isArray(raw.objectPermissions)) {
    out.objectPermissions = [...raw.objectPermissions].sort((a, b) => String(a.object).localeCompare(String(b.object)));
  }
  return out;
}

export function stripNoiseFields<T extends Record<string, any>>(input: T): Partial<T> {
  const out: any = {};
  for (const [k, v] of Object.entries(input)) if (!NOISE_FIELDS.has(k)) out[k] = v;
  return out;
}

import { canonicalizeField, canonicalizePermissionSet } from './drift-canonical';
import type { FieldDelta, PermissionDelta, DriftDelta, DriftEventPayload } from '@sandforge/shared';
import type { MetricBus } from '../monitor/MetricBus';

export class DriftDetector {
  // ...existing constructor + detect()...

  detectFieldDrift(prev: ObjectDescribe, curr: ObjectDescribe): FieldDelta[] {
    const objectApiName = curr.name;
    const prevFields = new Map<string, any>((prev.fields ?? []).map(f => [f.name, canonicalizeField(f)]));
    const currFields = new Map<string, any>((curr.fields ?? []).map(f => [f.name, canonicalizeField(f)]));
    const out: FieldDelta[] = [];
    const detectedAt = new Date().toISOString();

    // Added
    for (const name of currFields.keys()) {
      if (!prevFields.has(name)) out.push({ objectApiName, fieldApiName: name, changeKind: 'added', after: currFields.get(name), detectedAt });
    }
    // Removed
    for (const name of prevFields.keys()) {
      if (!currFields.has(name)) out.push({ objectApiName, fieldApiName: name, changeKind: 'removed', before: prevFields.get(name), detectedAt });
    }
    // Modified — use microdiff on the canonical projection
    for (const [name, currField] of currFields) {
      const prevField = prevFields.get(name);
      if (!prevField) continue;
      const changes = diff(prevField, currField);
      if (changes.length === 0) continue;
      // Classify
      for (const c of changes) {
        const kind = c.path[0] === 'type' ? 'type-changed'
                   : c.path[0] === 'length' ? 'length-changed'
                   : c.path[0] === 'picklistValues' ? 'picklist-changed'
                   : c.path[0] === 'required' ? 'required-changed'
                   : 'type-changed';
        out.push({ objectApiName, fieldApiName: name, changeKind: kind as any, before: prevField, after: currField, detectedAt });
      }
    }
    return out;
  }

  detectPermissionDrift(prev: PermissionSet | Profile, curr: PermissionSet | Profile): PermissionDelta[] {
    const profileOrPermSetName = curr.name;
    const cPrev = canonicalizePermissionSet(prev);
    const cCurr = canonicalizePermissionSet(curr);
    const detectedAt = new Date().toISOString();
    const out: PermissionDelta[] = [];

    const indexFP = (perms: any[] = []) => new Map(perms.map(p => [`${p.field}`, p]));
    const prevFP = indexFP(cPrev.fieldPermissions);
    const currFP = indexFP(cCurr.fieldPermissions);

    for (const [field, currPerm] of currFP) {
      const prevPerm = prevFP.get(field);
      if (!prevPerm) {
        out.push({ profileOrPermSetName, fieldApiName: field, changeKind: 'granted', permission: 'read', after: !!currPerm.read, detectedAt });
        continue;
      }
      for (const k of ['read', 'edit'] as const) {
        if (Boolean(prevPerm[k]) !== Boolean(currPerm[k])) {
          out.push({ profileOrPermSetName, fieldApiName: field, changeKind: currPerm[k] ? 'granted' : 'revoked', permission: k, before: !!prevPerm[k], after: !!currPerm[k], detectedAt });
        }
      }
    }
    // Mirror logic for revoked-only fields + objectPermissions
    return out;
  }

  /**
   * High-level entry: scan a snapshot pair, debounce per (orgId, pairId), emit on bus.
   * @param orgId
   * @param snapshotPairId stable id of the pair, e.g. `${prev.id}->${curr.id}`
   * @param prevPayload list of describes from the previous snapshot
   * @param currPayload list of describes from the current snapshot
   */
  scanAndEmit(orgId: string, snapshotPairId: string, prevPayload: SnapshotPayload, currPayload: SnapshotPayload, bus: MetricBus): DriftEventPayload | null {
    if (this.recentEmits.has(snapshotPairId)) {
      const lastEmit = this.recentEmits.get(snapshotPairId)!;
      if (Date.now() - lastEmit < 60_000) return null;     // P-03.2 debounce
    }
    const fieldDeltas: DriftDelta[] = [];
    const permDeltas: DriftDelta[] = [];
    // ...iterate prev/curr objects, call detectFieldDrift + detectPermissionDrift...
    const allDeltas: DriftDelta[] = [...fieldDeltas, ...permDeltas];

    // P-03.2 threshold filter: skip if < 1% of fields touched on a CustomObject
    const significantDeltas = this.filterSignificant(allDeltas, currPayload);
    if (significantDeltas.length === 0) return null;

    const severity: 'info' | 'breaking' | 'permission' =
      permDeltas.length > 0 ? 'permission'
      : fieldDeltas.some(d => d.changeKind === 'removed' || d.changeKind === 'type-changed') ? 'breaking'
      : 'info';

    const payload: DriftEventPayload = {
      orgId, snapshotPairId,
      summary: `${significantDeltas.length} drift change(s) detected`,
      deltaCount: significantDeltas.length,
      severity,
      deltas: significantDeltas.slice(0, 500),
    };

    bus.emit('monitor:drift:detected', payload as any);   // shape match per Plan 03-01 envelope
    this.recentEmits.set(snapshotPairId, Date.now());
    return payload;
  }
